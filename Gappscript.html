function doGet() { try { const data = getMasterData(); return ContentService.createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); } catch (e) { return ContentService.createTextOutput(JSON.stringify({error: e.toString()})) .setMimeType(ContentService.MimeType.JSON); } } function getMasterData() { const ss = SpreadsheetApp.getActiveSpreadsheet(); const hotelSheet = ss.getSheetByName("Master-Hotels"); const sightSheet = ss.getSheetByName("Master-Sightseeing"); if (!hotelSheet || !sightSheet) { throw new Error("Sheet names 'Master-Hotels' or 'Master-Sightseeing' not found."); } const hotels = hotelSheet.getDataRange().getValues().slice(1); const sights = sightSheet.getDataRange().getValues().slice(1); return { hotels: hotels.filter(r => r[0]).map(r => ({ city: String(r[0]).trim(), name: String(r[1]).trim(), type: String(r[2]).trim(), cost: parseFloat(String(r[3]).replace(/[^\d.]/g, "")) || 0 })), sights: sights.filter(r => r[0]).map(r => ({ info: String(r[0]).trim(), // Col A: Information city: String(r[1]).trim(), // Col B: City price: parseFloat(String(r[2]).replace(/[^\d.]/g, "")) || 0 // Col C: Price })) }; }
